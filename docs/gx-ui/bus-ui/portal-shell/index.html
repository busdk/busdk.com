Bus UI Portal Integration

PortalShell renders host-aware module pages.

Use PortalShell when a product module needs shared portal chrome while the host still owns base paths, public config, assets, and session context.

Public API

Functionuiportal.PortalShell
Propsuiportal.PortalShellProps

Go API

import (
  gx "github.com/busdk/bus-gx/pkg/gx"
  uiportal "github.com/busdk/bus-ui/pkg/uiportal"
)

node := uiportal.PortalShell(uiportal.PortalShellProps{
  Title: "Accounting",
  HostContext: uiportal.HostContext{
    ModuleID: "accounting",
    BasePath: "/modules/accounting/",
  },
  MainNodes: []gx.Node{
    gx.Element("p", nil, gx.Text("Portal content")),
  },
})

package portalui

import gx "github.com/busdk/bus-gx/pkg/gx"

func PortalShellExample() gx.Node {

    <main><Text value={"Portal content"}></Text></main>
