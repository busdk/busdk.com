GX Nodes

VNode preserves a render tree.

Use VNode helpers when you need to inspect or transform element, text, and trusted raw nodes before rendering.

Public API

Typeui.VNode
Functionsui.VEl, ui.VText, ui.VRaw

Go API

import ui "github.com/busdk/bus-ui/pkg/ui"

tree := ui.VEl("article", ui.P("class", "summary"),
  ui.VEl("h2", nil, ui.VText("Virtual node")),
)
html := tree.HTML()

package nodes

import gx "github.com/busdk/bus-gx/pkg/gx"

func Summary() gx.Node {
  return <article class="summary">
    <h2><Text value={"Virtual node"}></Text></h2>
  </article>
}
